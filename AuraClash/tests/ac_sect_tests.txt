		scope:ac_test_agent = {
			set_variable = { name = ac_sect_contribution value = 0 }
			set_variable = { name = ac_sect_merit value = 0 }
            # Keep the encounter asynchronous: a local NPC job resolves its
            # choices immediately, before the explicit reward assertion below.
            create_task_contract = {
                task_contract_type = ac_sect_patrol task_contract_tier = 1
                task_contract_employer = var:ac_sect.story_owner
                location = title:k_scotland.var:ac_kingdom_sect.var:ac_headquarters

		scope:ac_test_agent.var:ac_sect_mission = { complete_task_contract = ac_sect_patrol_success }

ac_sect_test_06_relationships_and_succession = {
	title_target = k_england
	assert = { has_variable = ac_kingdom_sect title:k_scotland = { exists = holder } }
	effect = {
		var:ac_kingdom_sect = { save_scope_as = ac_inherited_sect var:ac_estate_title.title_domicile = { save_scope_as = ac_inherited_domicile } story_owner = { save_scope_as = actor } }
		title:k_scotland.var:ac_kingdom_sect.story_owner = { save_scope_as = recipient }
		ac_sect_alliance_effect = yes
		var:ac_kingdom_sect = { save_scope_as = ac_join_sect }
		title:k_scotland.holder = {
			save_scope_as = ac_test_ruler
			ac_sect_join_effect = yes
			remove_trait = ac_sect_lower_disciple
			add_trait = ac_sect_upper_disciple
			set_variable = { name = ac_sect_merit value = 50000 }
		}
		var:ac_kingdom_sect = {
			random_in_list = {
				variable = ac_members
				limit = { has_trait = ac_sect_lower_disciple }
				set_variable = { name = ac_sect_merit value = 100000 }
			}
			story_owner = { death = natural }
		}
	}
	expect = {
		var:ac_kingdom_sect = scope:ac_inherited_sect
		var:ac_kingdom_sect.story_owner = scope:ac_test_ruler
		var:ac_kingdom_sect.var:ac_master = scope:ac_test_ruler
		var:ac_kingdom_sect.var:ac_alchemy_pavilion = 0
        var:ac_kingdom_sect.var:ac_estate_title = {
            holder = scope:ac_test_ruler
            title_domicile = scope:ac_inherited_domicile
            title_domicile = { owner = scope:ac_test_ruler has_domicile_construction = ac_sect_alchemy_pavilion_01 }
        }
		var:ac_kingdom_sect = { is_target_in_variable_list = { name = ac_allies target = scope:recipient.var:ac_sect } }
		scope:recipient.var:ac_sect = { is_target_in_variable_list = { name = ac_allies target = scope:ac_inherited_sect } }
		scope:ac_test_ruler = { has_trait = ac_sect_master primary_title = title:k_scotland has_government = feudal_government }
	}

ac_sect_test_10_native_trial_admission_and_promotion = {
	title_target = k_england
	assert = { any_activity = { has_activity_type = activity_ac_sect_trials var:ac_trial_sect ?= title:k_england.var:ac_kingdom_sect } }
	effect = {
		random_activity = {
			limit = { has_activity_type = activity_ac_sect_trials var:ac_trial_sect ?= title:k_england.var:ac_kingdom_sect }
			save_scope_as = activity
			activity_host = { save_scope_as = host }
		}
		holder = {
			create_character = {
				gender_female_chance = 50
				age = 25
				culture = this.culture faith = this.faith location = scope:activity.activity_location
				save_scope_as = ac_test_examinee
			}
		}
		scope:ac_test_examinee = {
			add_to_activity_without_travel = scope:activity
			ac_sect_begin_trial_effect = yes
			set_variable = { name = ac_sect_trial_score value = 100 }
			ac_sect_complete_trial_effect = yes
		}
	}
	expect = { scope:ac_test_examinee = { involved_activity = scope:activity has_trait = ac_cultivator has_trait = ac_sect_lower_disciple var:ac_sect = title:k_england.var:ac_kingdom_sect } }
	effect = {
		scope:ac_test_examinee = {
			remove_character_flag = ac_sect_trial_cooldown
			set_variable = { name = ac_sect_contribution value = 200 }
			set_variable = { name = ac_sect_promotion_requested value = 1 }
			ac_sect_begin_trial_effect = yes
		}
	}
	assert = { scope:ac_test_examinee = { var:ac_sect_contribution = 50 has_variable = ac_sect_trial_paid } }
	effect = {
		scope:ac_test_examinee = {
			set_variable = { name = ac_sect_trial_score value = 100 }
			ac_sect_complete_trial_effect = yes
			ac_sect_complete_trial_effect = yes
		}
	}
	expect = {
		scope:ac_test_examinee = {
			has_trait = ac_sect_upper_disciple NOT = { has_trait = ac_sect_lower_disciple }
			var:ac_sect_contribution = 50 NOT = { has_variable = ac_sect_trial_paid }
			ac_element_count = 2
		}
	}
}

# A Master may already have a personal native estate. The sect office must not
# replace it, change their government, or take away the imperial title.
ac_sect_test_14_personal_estate_coexistence = {
	title_target = k_wales
	assert = { title:e_byzantium.holder = { has_domicile = yes has_government = administrative_government } }
	effect = {
		var:ac_kingdom_sect = { save_scope_as = ac_join_sect story_owner = { save_scope_as = ac_returning_master } }
		title:e_byzantium.holder = {
			save_scope_as = ac_personal_estate_master
			domicile = { save_scope_as = ac_personal_estate }
			ac_sect_join_effect = yes
			remove_trait = ac_sect_lower_disciple add_trait = ac_sect_master
		}
		scope:ac_returning_master = { remove_trait = ac_sect_master add_trait = ac_sect_upper_disciple }
		var:ac_kingdom_sect = {
			make_story_owner = scope:ac_personal_estate_master
			set_variable = { name = ac_master value = scope:ac_personal_estate_master }
			ac_sect_ensure_domicile_effect = yes
		}
	}
	expect = {
		scope:ac_personal_estate_master = { primary_title = title:e_byzantium has_government = administrative_government }
		scope:ac_personal_estate = { owner = scope:ac_personal_estate_master }
		var:ac_kingdom_sect.var:ac_estate_title = {
			holder = scope:ac_personal_estate_master
			title_domicile = { owner = scope:ac_personal_estate_master is_domicile_type = ac_sect_estate this != scope:ac_personal_estate }
		}
	}
	effect = {
		scope:ac_personal_estate_master = { remove_trait = ac_sect_master add_trait = ac_sect_upper_disciple }
		scope:ac_returning_master = { remove_trait = ac_sect_upper_disciple add_trait = ac_sect_master }
		var:ac_kingdom_sect = {
			make_story_owner = scope:ac_returning_master
			set_variable = { name = ac_master value = scope:ac_returning_master }
			ac_sect_ensure_domicile_effect = yes
			ac_sect_recount_effect = yes
		}
		scope:ac_personal_estate_master = { ac_sect_leave_effect = yes }
	}
	expect = {
		scope:ac_personal_estate_master = { primary_title = title:e_byzantium has_government = administrative_government }
		scope:ac_personal_estate = { owner = scope:ac_personal_estate_master }
		var:ac_kingdom_sect.var:ac_estate_title.holder = scope:ac_returning_master
	}
}

# Migration preserves an already paid legacy project while completed levels
# become native buildings, then finishes the old project exactly once.
ac_sect_test_15_legacy_construction_migration = {
    title_target = k_wales
    effect = {
        title:k_wales.var:ac_kingdom_sect = {
            set_variable = { name = ac_treasury value = 900 }
            var:ac_estate_title.title_domicile = { remove_domicile_building_no_refund = ac_sect_training_ground_01 }
            set_variable = { name = ac_training_ground value = 1 }
            set_variable = { name = ac_building value = flag:scripture_library }
            set_variable = { name = ac_building_timer value = 1 days = 180 }
            ac_sect_migrate_building_levels_effect = yes
        }
    }
    expect = {
        title:k_wales.var:ac_kingdom_sect = {
            var:ac_treasury = 900
        }
    }
    expect = { title:k_wales.var:ac_kingdom_sect = { has_variable = ac_building_timer var:ac_building = flag:scripture_library } }
    expect = { title:k_wales.var:ac_kingdom_sect.var:ac_estate_title.title_domicile = { has_domicile_building = ac_sect_training_ground_01 } }
    effect = {
        title:k_wales.var:ac_kingdom_sect = {
            remove_variable = ac_building_timer
            ac_sect_finish_construction_effect = yes
            ac_sect_finish_construction_effect = yes
        }
    }
    expect = {
        title:k_wales.var:ac_kingdom_sect = {
            var:ac_treasury = 900
        }
    }
    expect = { title:k_wales.var:ac_kingdom_sect = { var:ac_scripture_library = 1 NOT = { has_variable = ac_building } } }
    expect = { title:k_wales.var:ac_kingdom_sect.var:ac_estate_title.title_domicile = { has_domicile_building = ac_sect_scripture_library_01 } }
}
