ac_sect.0010 = {
	type = character_event
	title = ac_sect.0010.t
	desc = ac_sect.0010.desc
	theme = stewardship
	trigger = { ac_sect_member_trigger = yes }
	option = {
		name = ac_sect.0010.pill
		trigger = { var:ac_sect_contribution >= 50 var:ac_sect.var:ac_alchemy_pavilion >= 1 }
		change_variable = { name = ac_sect_contribution subtract = 50 }
		if = { limit = { NOT = { has_variable = ac_sect_pills } } set_variable = { name = ac_sect_pills value = 0 } }
		change_variable = { name = ac_sect_pills add = 1 }
		custom_tooltip = ac_sect_pill_tt
	}
	option = {
		name = ac_sect.0010.manual
		trigger = {
			var:ac_sect_contribution >= 75
			var:ac_sect.var:ac_scripture_library >= 1
			NOT = { has_character_flag = ac_sect_manual_cooldown }
		}
		change_variable = { name = ac_sect_contribution subtract = 75 }
		add_ac_cultivation_lifestyle_xp = 150
		add_character_flag = { flag = ac_sect_manual_cooldown months = 6 }
	}
	option = {
		name = ac_sect.0010.gold
		trigger = { var:ac_sect_contribution >= 50 var:ac_sect.var:ac_treasury >= 40 }
		change_variable = { name = ac_sect_contribution subtract = 50 }
		var:ac_sect = { change_variable = { name = ac_treasury subtract = 40 } }
		add_gold = 40
	}
	option = { name = ac_sect_back trigger_event = ac_sect.0001 }
}

ac_sect.0021 = {
	type = character_event
	title = ac_sect.0021.t
	desc = ac_sect.0021.desc
	theme = diplomacy
	trigger = { ac_sect_member_trigger = yes has_trait = ac_sect_upper_disciple var:ac_sect_contribution >= 200 }
	option = {
		name = ac_sect.0021.petition
		change_variable = { name = ac_sect_contribution subtract = 200 }
		random_list = {
			60 = {
				var:ac_sect.story_owner = { save_scope_as = actor }
				save_scope_as = recipient
				scope:actor = { remove_trait = ac_sect_master add_trait = ac_sect_upper_disciple }
				remove_trait = ac_sect_upper_disciple
				add_trait = ac_sect_master
				var:ac_sect = { make_story_owner = scope:recipient set_variable = { name = ac_master value = scope:recipient } ac_sect_ensure_domicile_effect = yes ac_sect_recount_effect = yes }
				trigger_event = ac_sect.0020
			}
			40 = { add_stress = 20 }
		}
	}
	option = { name = ac_sect_close }
}
