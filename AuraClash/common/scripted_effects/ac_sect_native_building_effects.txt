ac_sect_build_cultivation_hall_effect = {
	if = {
		limit = {
			ac_sect_master_trigger = yes
			var:ac_sect = { NOT = { has_variable = ac_building } var:ac_cultivation_hall < 3 exists = var:ac_estate_title.title_domicile }
			var:ac_sect.var:ac_treasury >= ac_sect_cultivation_hall_cost
			var:ac_sect.var:ac_estate_title.title_domicile = { NOT = { has_ongoing_domicile_construction = yes } }
		}
		var:ac_sect.var:ac_estate_title.title_domicile = {
			if = {
				limit = { var:ac_domicile_sect.var:ac_cultivation_hall = 0 }
				construct_domicile_building = ac_sect_cultivation_hall_01
			}
			else_if = {
				limit = { var:ac_domicile_sect.var:ac_cultivation_hall = 1 }
				construct_domicile_building = ac_sect_cultivation_hall_02
			}
			else_if = {
				limit = { var:ac_domicile_sect.var:ac_cultivation_hall = 2 }
				construct_domicile_building = ac_sect_cultivation_hall_03
			}
		}
	}
}
