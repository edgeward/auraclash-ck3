# Cultivation event techniques. Travel helpers must actually be in the entourage.
ac_event_practitioner_trigger = {
	has_trait = ac_cultivator
	age >= 16
	is_alive = yes
	is_imprisoned = no
	NOT = { has_trait = incapable }
}

ac_event_available_practitioner_trigger = {
	OR = {
		ac_event_$ABILITY$_trigger = yes
		AND = {
			exists = current_travel_plan
			current_travel_plan = {
				any_entourage_character = { ac_event_$ABILITY$_trigger = yes }
			}
		}
	}
}
