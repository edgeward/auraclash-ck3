	effect = {
		ac_sect_save_activity_home_effect = yes
		scope:ac_activity_home = { every_in_list = { variable = ac_members limit = { is_alive = yes } add_to_list = characters } }
	}
}

		ac_sect_save_activity_home_effect = yes
		scope:ac_activity_home = {
			every_in_list = {
				variable = ac_allies
				every_in_list = { variable = ac_members limit = { is_alive = yes } add_to_list = characters }
			}
		}
	}
}

		}
	}
