ac_sect_native_estate_info = {
	scope = character
	is_shown = { ac_sect_member_trigger = yes has_variable = ac_sect_estate_open }
	effect = { remove_variable = ac_sect_estate_open remove_variable = ac_sect_estate_selected }
}
ac_sect_estate_cancel = { scope = character
	is_shown = { ac_sect_master_trigger = yes }
	is_valid = { ac_sect_master_trigger = yes var:ac_sect.var:ac_estate_domicile = { has_ongoing_domicile_construction = yes } }
	effect = { hidden_effect = { var:ac_sect.var:ac_estate_domicile = { ac_sect_cancel_native_construction_effect = yes } } }
}
ac_sect_estate_close_panel = { scope = character is_shown = { ac_sect_member_trigger = yes } effect = { remove_variable = ac_sect_estate_selected } }
ac_sect_estate_select_cultivation_hall = { scope = character
	is_shown = { ac_sect_member_trigger = yes }
	effect = { set_variable = { name = ac_sect_estate_selected value = flag:cultivation_hall } }
}
ac_sect_estate_select_cultivation_hall_panel = { scope = character is_shown = { ac_sect_member_trigger = yes var:ac_sect_estate_selected ?= flag:cultivation_hall } }
ac_sect_estate_commission_cultivation_hall = { scope = character
	is_shown = { ac_sect_master_trigger = yes }
	is_valid = {
		ac_sect_master_trigger = yes
		var:ac_sect = { NOT = { has_variable = ac_building } var:ac_cultivation_hall < 3 }
		var:ac_sect.var:ac_treasury >= ac_sect_cultivation_hall_cost
		var:ac_sect.var:ac_estate_domicile = { NOT = { has_ongoing_domicile_construction = yes } }
	}
	effect = { ac_sect_build_cultivation_hall_effect = yes }
}
ac_sect_estate_pending_cultivation_hall = { scope = character
	is_shown = { var:ac_sect = { has_variable = ac_native_construction var:ac_building ?= flag:cultivation_hall } }
}
